    \begin{aligned}

    &=-\frac{\operatorname{sen}^{n-1}x\cos^{n+1}x}{m+n}+\frac{n-1}{m+n}\int\cos^mx\operatorname{sen}^{n-2}x\,dx
    \end{aligned}
    $$
